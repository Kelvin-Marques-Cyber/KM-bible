cmake_minimum_required(VERSION 3.22.1)
project(needle_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(needle_core STATIC IMPORTED)
set_target_properties(needle_core PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../../../native/needle/android-arm64/libneedle.a")
add_library(needle_runtime SHARED src/main/cpp/runtime.cpp)
target_include_directories(needle_runtime PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../native/needle/android-arm64")
target_link_libraries(needle_runtime PRIVATE needle_core log android)
target_link_options(needle_runtime PRIVATE "-Wl,-z,max-page-size=16384")
